function doGet(e) {
  const sheet = SpreadsheetApp
    .openById('1eBX-OiTGma5nCnmx3kK3GQziUPpe269Lr9oVjIiVgWU')
    .getSheetByName('Form Responses');

  if (e.parameter.action === 'checkDuplicate') {
    const result = findDuplicate(sheet, e.parameter.invoice || '');

    return ContentService
      .createTextOutput(JSON.stringify(result))
      .setMimeType(ContentService.MimeType.JSON);
  }

  if (e.parameter.action === 'updateInvoiceRow') {
    const updated = withLock(() => updateInvoiceRow(
      sheet,
      e.parameter.invoice || '',
      e.parameter.customer || '',
      Number(e.parameter.total || 0)
    ));

    return ContentService
      .createTextOutput(JSON.stringify({ updated: updated }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  // Default / legacy path: submit a new order. Re-checks for a duplicate
  // right before writing (inside the same lock as the write itself), since
  // the client's own upfront checkDuplicate call happened as a separate,
  // earlier request and could have raced with another submission for the
  // same invoice in the meantime.
  const result = withLock(() => submitOrder(sheet, e.parameter));

  return ContentService
    .createTextOutput(JSON.stringify(result))
    .setMimeType(ContentService.MimeType.JSON);
}

// Serializes access to the block passed in, so two concurrent doGet
// executions can never interleave a duplicate check with a write. Waits up
// to 30s for another execution to finish before giving up.
function withLock(fn) {
  const lock = LockService.getScriptLock();
  lock.waitLock(30000);
  try {
    return fn();
  } finally {
    lock.releaseLock();
  }
}

function submitOrder(sheet, params) {
  const invoice = params.invoice || '';
  const existingRow = findMatchingRow(sheet, invoice);
  if (existingRow !== null) {
    const rowValues = sheet.getRange(existingRow, 1, 1, 7).getValues()[0]; // columns A-G
    return {
      status: 'duplicate',
      existingCustomer: String(rowValues[2] || ''),
      existingTotal: Number(rowValues[6]) || 0,
    };
  }

  sheet.appendRow([
    invoice,
    params.state || '',
    params.customer || '',
    params.edits || '',
    params.submittedBy || '',
    params.dateUtc || '',
    Number(params.total || 0),
    params.payTo || '',
    params.license || '',
  ]);

  return { status: 'ok' };
}

// Only scans the most recent MAX_DUPLICATE_CHECK_ROWS rows instead of the
// whole sheet, since legitimate duplicate resubmissions only happen shortly
// after the original entry.
const MAX_DUPLICATE_CHECK_ROWS = 7000;

// Column A = invoice. Compares case/whitespace-insensitively. Searches
// backward so ties resolve to the most recent row. Shared by findDuplicate,
// submitOrder and updateInvoiceRow.
function findMatchingRow(sheet, invoice) {
  const normalizedInvoice = invoice.trim().toLowerCase();

  if (!normalizedInvoice) {
    return null;
  }

  const lastRow = sheet.getLastRow();
  if (lastRow < 1) {
    return null;
  }

  const startRow = Math.max(1, lastRow - MAX_DUPLICATE_CHECK_ROWS + 1);
  const numRows = lastRow - startRow + 1;
  const values = sheet.getRange(startRow, 1, numRows, 1).getValues(); // column A

  for (let i = values.length - 1; i >= 0; i--) {
    const rowInvoice = String(values[i][0]).trim().toLowerCase();
    if (rowInvoice === normalizedInvoice) {
      return startRow + i;
    }
  }

  return null;
}

// Returns { duplicate: false } or { duplicate: true, existingCustomer,
// existingTotal } where existingCustomer/existingTotal are the customer
// name (column C) and Dollar Amount (column G) currently on the matched row.
function findDuplicate(sheet, invoice) {
  const row = findMatchingRow(sheet, invoice);
  if (row === null) {
    return { duplicate: false };
  }

  const rowValues = sheet.getRange(row, 1, 1, 7).getValues()[0]; // columns A-G
  return {
    duplicate: true,
    existingCustomer: String(rowValues[2] || ''),
    existingTotal: Number(rowValues[6]) || 0,
  };
}

// Overwrites the customer name (column C) and Dollar Amount (column G) on
// the most recent matching row. Returns true if a row was updated, false if
// no match was found.
function updateInvoiceRow(sheet, invoice, customer, total) {
  const row = findMatchingRow(sheet, invoice);
  if (row === null) {
    return false;
  }

  sheet.getRange(row, 3).setValue(customer);
  sheet.getRange(row, 7).setValue(total);
  return true;
}
